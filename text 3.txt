server/server.js
require("dotenv").config(); const express = require("express"); const mongoose = require("mongoose"); const cors = require("cors"); const bcrypt = require("bcrypt"); const app = express(); app.use(cors()); app.use(express.json()); mongoose.connect(process.env.MONGO_URI || "your-mongo-here"); const User = mongoose.model("User", new mongoose.Schema({name:String,phone:{type:String,unique:true},password:String,type:String})); app.post("/api/register", async (req, res) => { const { name, phone, password, type } = req.body; const hashed = await bcrypt.hash(password, 10); try { await User.create({ name, phone, password: hashed, type }); res.json({success:true}); } catch (e) { res.status(400).json({error:"Phone exists"}); }}); app.listen(process.env.PORT || 5000, () => console.log("Kpon Kpor running"));
